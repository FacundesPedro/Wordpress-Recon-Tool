# WHOIS patterns for .br (Brazilian domains)
# Format: field_name:regex_pattern
# Uses re.MULTILINE | re.IGNORECASE flags
# Brazilian WHOIS uses different field names (nserver, owner, etc.)
nserver:^nserver:\s*(.+)
owner:^owner:\s*(.+)
responsible:^responsible:\s*(.+)
changed:^changed:\s*(.+)
created:^created:\s*(.+)
expires:^expires:\s*(.+)
status:^status:\s*(.+)
country:^country:\s*(.+)
admin_c:^admin-c:\s*(.+)
tech_c:^tech-c:\s*(.+)
nic_handle:^nic-hdl-br:\s*(.+)
email:^e-?mail:\s*(.+)
whois:^whois:\s*(.+)
dnssec:DNSSEC:\s*(.+)
ds_rdata:^ds-rdata:\s*(.+)
